Busca productos de compra en la empresa. Utilizado para autocompletar la selección de productos al crear o filtrar registros de otros recursos (órdenes de compra, recepciones, entregas, facturas por pagar).

<Note>
  Este endpoint requiere el scope **`PurchaseProducts.Read`**.
</Note>

### Headers

<ParamField header="X-Tenant-Id" type="string" required>
  Identificador del tenant al cual pertenece el API Key.
</ParamField>

### Query parameters

<ParamField query="skip" default="0" type="integer">
  Número de registros a omitir desde el inicio de los resultados. Utilízado junto con `take` para paginar los resultados.
</ParamField>

<ParamField query="take" default="20" type="integer">
  Número de registros a retornar. Por defecto `20`, máximo `100`.
</ParamField>

<ParamField query="Name" type="StringSearchFilter">
  Filtra por el nombre del producto de compra.
</ParamField>

<ParamField query="GoodServiceApplicability.Id" type="EnumSearchFilter">
  Filtra por la aplicabilidad del producto de compra. Valores aceptados:

  - `NotApplicable`
  - `Product`
  - `Service`
</ParamField>

<ParamField query="TaxRate" type="RangeSearchFilter<decimal>">
  Filtra por la tasa de impuesto del producto de compra.
</ParamField>

### Example request

<CodeGroup>

```bash curl
curl --request GET \
  --url "https://api.primenetcloud.com/v1/purchase-products/search?Name=Aceite*&skip=0&take=20" \
  --header "X-Api-Key: YOUR_API_KEY" \
  --header "X-Tenant-Id: TENANT_ID"
```

```javascript JavaScript
const response = await fetch(
  "https://api.primenetcloud.com/v1/purchase-products/search?Name=Aceite*&skip=0&take=20",
  {
    method: "GET",
    headers: {
      "X-Api-Key": "YOUR_API_KEY",
      "X-Tenant-Id": "TENANT_ID"
    },
  }
);

const data = await response.json();
console.log(data);
```

```python Python
import requests

response = requests.get(
    "https://api.primenetcloud.com/v1/purchase-products/search",
    params={
        "Name": "Aceite*",
        "skip": 0,
        "take": 20,
    },
    headers={
      "X-Api-Key": "YOUR_API_KEY",
      "X-Tenant-Id": "TENANT_ID"
    }
)

print(response.json())
```

</CodeGroup>

### Response fields

<ResponseField name="totalCount" type="integer">
  Total de registros.
</ResponseField>

<ResponseField name="data" type="array">
  Lista de objetos PurchaseProduct.

  <Expandable title="PurchaseProduct">
    <ResponseField name="id" type="integer">
      Identificador único del producto de compra.
    </ResponseField>

    <ResponseField name="goodServiceApplicability" type="object">
      Aplicabilidad del producto de compra.

      <Expandable title="propiedades">
        <ResponseField name="id" type="integer">
          Identificador de la aplicabilidad.

          Valores posibles:
          - `NotApplicable`
          - `Product`
          - `Service`
        </ResponseField>

        <ResponseField name="name" type="string">
          Nombre traducido de la aplicabilidad.
        </ResponseField>
      </Expandable>
    </ResponseField>

    <ResponseField name="name" type="string">
      Nombre del producto de compra.
    </ResponseField>

    <ResponseField name="category" type="object">
      Categoría de bienes/servicios del producto de compra.

      <Expandable title="propiedades">
        <ResponseField name="name" type="string">
          Nombre de la categoría.
        </ResponseField>

        <ResponseField name="account" type="object">
          Cuenta contable asociada a la categoría.

          <Expandable title="propiedades">
            <ResponseField name="useDepartment" type="object">
              Uso de departamento en la cuenta contable.

              <Expandable title="propiedades">
                <ResponseField name="id" type="integer">
                  Identificador del uso.

                  Valores posibles:
                  - `Mandatory`
                  - `Optional`
                  - `Disabled`
                </ResponseField>

                <ResponseField name="name" type="string">
                  Nombre traducido del uso.
                </ResponseField>
              </Expandable>
            </ResponseField>

            <ResponseField name="useCostCenter" type="object">
              Uso de centro de costo en la cuenta contable. Mismas propiedades que `useDepartment`.
            </ResponseField>

            <ResponseField name="useProject" type="object">
              Uso de proyecto en la cuenta contable. Mismas propiedades que `useDepartment`.
            </ResponseField>
          </Expandable>
        </ResponseField>
      </Expandable>
    </ResponseField>

    <ResponseField name="taxRate" type="decimal">
      Tasa de impuesto aplicada al producto de compra.
    </ResponseField>
  </Expandable>
</ResponseField>

### Errores

<ResponseField name="403 Forbidden">
  El API Key no tiene el scope `PurchaseProducts.Read` requerido.
</ResponseField>
