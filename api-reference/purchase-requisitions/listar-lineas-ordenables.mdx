---
title: "Líneas Ordenables"
api: "GET /purchase-requisitions/{id}/orderable-lines"
---

Obtén las líneas de cotización aprobadas de una requisición, con lo que queda pendiente por ordenar de cada una. Son las líneas que pueden enviarse a [Generar Órdenes](/api-reference/purchase-requisitions/generar-orden).

<Note>
  Este endpoint requiere el scope **`PurchaseRequisitions.Read`**.
</Note>

### Headers

<ParamField header="X-Tenant-Id" type="string" required>
  Identificador del tenant al cual pertenece el API Key.
</ParamField>

### Path parameters

<ParamField path="id" type="integer" required>
  Identificador único de la requisición de compra.
</ParamField>

### Query parameters

<ParamField query="vendor_id" type="integer">
  Filtra por el proveedor de la cotización.
</ParamField>

### Response fields

<ResponseField name="id" type="integer">
  Identificador de la línea de cotización aprobada. Es el valor a enviar en `details[].quoteDetailId` al generar órdenes.
</ResponseField>

<ResponseField name="quote" type="object">
  Cotización a la que pertenece la línea.

  <Expandable title="Ver propiedades">
    <ResponseField name="id" type="integer">
      Identificador de la cotización.
    </ResponseField>

    <ResponseField name="number" type="string">
      Número de la cotización asignado por el proveedor.
    </ResponseField>

    <ResponseField name="vendor" type="object">
      Proveedor.

      <Expandable title="Ver propiedades">
        <ResponseField name="id" type="integer">
          Identificador del proveedor.
        </ResponseField>

        <ResponseField name="name" type="string">
          Nombre del proveedor.
        </ResponseField>
      </Expandable>
    </ResponseField>

    <ResponseField name="currency" type="object">
      Moneda de la cotización.

      <Expandable title="Ver propiedades">
        <ResponseField name="id" type="string">
          Identificador de la moneda (ej. `DOP`, `USD`).
        </ResponseField>
      </Expandable>
    </ResponseField>

    <ResponseField name="paymentTerm" type="object">
      Término de pago de la cotización.

      <Expandable title="Ver propiedades">
        <ResponseField name="id" type="integer">
          Identificador del término de pago.
        </ResponseField>

        <ResponseField name="name" type="string">
          Nombre del término de pago.
        </ResponseField>
      </Expandable>
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="requisitionDetail" type="object">
  Línea de la requisición cotizada.

  <Expandable title="Ver propiedades">
    <ResponseField name="id" type="integer">
      Identificador de la línea.
    </ResponseField>

    <ResponseField name="lineNumber" type="integer">
      Número de la línea.
    </ResponseField>

    <ResponseField name="item" type="object">
      Ítem solicitado. Para líneas fuera de catálogo solo `type` tiene valor.

      <Expandable title="Ver propiedades">
        <ResponseField name="type" type="object">
          Tipo de ítem.

          <Expandable title="Ver propiedades">
            <ResponseField name="id" type="string">
              Identificador del tipo de ítem.

              Valores posibles:
              - `Product`
              - `FixedAsset`
            </ResponseField>

            <ResponseField name="name" type="string">
              Nombre traducido del tipo de ítem.
            </ResponseField>
          </Expandable>
        </ResponseField>

        <ResponseField name="id" type="integer">
          Identificador del producto o activo fijo.
        </ResponseField>

        <ResponseField name="code" type="string">
          Siempre `null` en este endpoint.
        </ResponseField>

        <ResponseField name="name" type="string">
          Nombre del producto o activo fijo.
        </ResponseField>
      </Expandable>
    </ResponseField>

    <ResponseField name="description" type="string">
      Descripción de la línea fuera de catálogo.
    </ResponseField>

    <ResponseField name="isOutOfCatalog" type="boolean">
      Indica si la línea está fuera de catálogo.
    </ResponseField>

    <ResponseField name="unitName" type="string">
      Unidad en la que se expresa la cantidad.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="productPackaging" type="object">
  Empaque cotizado. Sus propiedades son `null` si no tiene.

  <Expandable title="Ver propiedades">
    <ResponseField name="id" type="integer">
      Identificador del empaque.
    </ResponseField>

    <ResponseField name="name" type="string">
      Nombre del empaque.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="packagingQuantity" type="decimal">
  Unidades por empaque de la cotización.
</ResponseField>

<ResponseField name="quantity" type="decimal">
  Cantidad aprobada, en empaques de la cotización.
</ResponseField>

<ResponseField name="approvedUnits" type="decimal">
  Unidades aprobadas (`quantity` × `packagingQuantity`).
</ResponseField>

<ResponseField name="orderedUnits" type="decimal">
  Unidades ya incluidas en órdenes de compra activas.
</ResponseField>

<ResponseField name="pendingUnits" type="decimal">
  Unidades aprobadas pendientes de ordenar.
</ResponseField>

<ResponseField name="price" type="decimal">
  Precio unitario cotizado.
</ResponseField>

<ResponseField name="discount" type="decimal">
  Descuento por unidad.
</ResponseField>

<ResponseField name="taxPercent" type="decimal">
  Porcentaje de impuesto cotizado.
</ResponseField>

<ResponseField name="isOrderable" type="boolean">
  Indica si la línea puede ordenarse: debe estar vinculada a un ítem del catálogo y tener unidades pendientes.
</ResponseField>

### Errores

<ResponseField name="404 Not Found">
  No existe ninguna requisición de compra con el `id` especificado en la URL.
</ResponseField>

<ResponseField name="403 Forbidden">
  El API Key no tiene el scope `PurchaseRequisitions.Read` requerido.
</ResponseField>
